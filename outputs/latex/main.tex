\section{背景与系统}

\begin{frame}[t]{背景与意义}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.31\textwidth}
      \begin{block}{设备问题}
        \begin{itemize}
          \item 轴承退化会引起振动、噪声和精度问题
          \item 严重时导致停机和设备损坏
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.31\textwidth}
      \begin{block}{RUL 预测}
        \begin{itemize}
          \item 回答“还能运行多久”
          \item 支持检修排期和备件准备
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.31\textwidth}
      \begin{block}{Fault 检测}
        \begin{itemize}
          \item 回答“现在是否异常”
          \item 用作早期报警和重点监控入口
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}

  \vspace{1.3em}
  \centering
  \begin{tikzpicture}[x=1cm,y=1cm]
    \node[draw=phmline, fill=phmlight, rounded corners, minimum width=3.2cm, minimum height=0.8cm] (a) at (0,0) {事后维修};
    \node[draw=phmline, fill=phmlight, rounded corners, minimum width=3.2cm, minimum height=0.8cm] (b) at (4.3,0) {定期维护};
    \node[draw=phmblue, fill=phmblue!8, rounded corners, very thick, minimum width=3.6cm, minimum height=0.8cm] (c) at (8.8,0) {\bfseries 预测性维护};

  \end{tikzpicture}

  \takeaway{本系统用轴承振动数据支撑两类决策：剩余寿命预测和故障状态判断。}
\end{frame}

\begin{frame}[t]{系统架构}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.36\textwidth}
      \begin{block}{四层架构}
        \begin{enumerate}
          \item 数据接入层：读取 PHM2012 与 XJTU-SY 原始振动信号
          \item 信号处理与特征层：窗口切分、频域变换、统计特征、标签构造
          \item 任务模型层：RUL 回归分支与 Fault 分类分支
          \item 训练评估层：训练、指标计算、曲线和结果输出
        \end{enumerate}
      \end{block}
      \takeaway{两个实验不是零散脚本，而是同一套 PHM 数据流的两条任务分支。}

\begin{frame}[t]{XJTU-SY 标签与特征}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.43\textwidth}
      \begin{block}{标签构造}
        \begin{itemize}
          \item 取每个轴承前 30\% 样本作为健康基线
          \item 计算 RMS 均值和标准差
          \item 后续样本连续超过 3$\sigma$ 阈值时，定义首次故障点 FOT
          \item FOT 前为 Healthy，FOT 后为 Faulty
        \end{itemize}
      \end{block}
      \begin{block}{输入形状}
        双通道合计 552 维，8 个快照组成序列：\texttt{[B, 8, 552]}。
      \end{block}
    \end{column}
    \begin{column}{0.54\textwidth}
      \centering
      \includegraphics[width=\linewidth,height=0.72\textheight,keepaspectratio]{xjtu_feature_analysis.png}
      \smallcap{图：XJTU-SY 特征组成}
      \smallskip
      {\color{phmblue}\bfseries RMS、频带能量和频谱分布用于刻画健康与故障边界。}
    \end{column}
  \end{columns}
\end{frame}
